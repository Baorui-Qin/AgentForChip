[BEGIN]
module TopModule (
    input  logic clk,
    input  logic x,
    output logic z
);

    logic q1 = 1'b0;
    logic q2 = 1'b0;
    logic q3 = 1'b0;

    always_ff @(posedge clk) begin
        q1 <= x ^ q1;
        q2 <= x & ~q2;
        q3 <= x | ~q3;
    end

    assign z = ~(q1 | q2 | q3);

endmodule
[DONE]
