    input  logic clk,
    input  logic resetn,
    input  logic x,
    input  logic y,
    output logic f,
    output logic g
);

    typedef enum logic [3:0] {
        A, B, C, D, E, F, G, H, I
    } state_t;

    state_t state, next_state;

    always_ff @(posedge clk) begin
        if (!resetn)
            state <= A;
        else
            state <= next_state;
    end

        next_state = state;
        case (state)

            G: next_state = y ? H : I;
            H: next_state = H;
            I: next_state = I;
