[BEGIN]
module TopModule (
    input clk,
    input resetn,
    input x,
    input y,
    output reg f,
    output reg g
);

    localparam [3:0] A     = 4'd0,
                     F1    = 4'd1,
                     S0    = 4'd2,
                     S1    = 4'd3,
                     S2    = 4'd4,
                     G1    = 4'd5,
                     G2    = 4'd6,
                     GPERM = 4'd7,
                     GZERO = 4'd8;

    reg [3:0] state, next;

    always @(posedge clk) begin
        if (!resetn)
            state <= A;
        else
            state <= next;
    end

    always @(*) begin
        f = 1'b0;
        g = 1'b0;
        next = state;

        case (state)
            A: begin
                if (resetn)
                    next = F1;
                else
                    next = A;
            end
            F1: begin
                f = 1'b1;
                next = S0;
            end
            S0: begin
                if (x)
                    next = S1;
                else
                    next = S0;
            end
            S1: begin
                if (x)
                    next = S1;
                else
                    next = S2;
            end
            S2: begin
                if (x)
                    next = G1;
                else
                    next = S0;
            end
            G1: begin
                g = 1'b1;
                if (y)
                    next = GPERM;
                else
                    next = G2;
            end
            G2: begin
                g = 1'b1;
                if (y)
                    next = GPERM;
                else
                    next = GZERO;
            end
            GPERM: begin
                g = 1'b1;
                next = GPERM;
            end
            GZERO: begin
                next = GZERO;
            end
            default: begin
                next = A;
            end
        endcase
    end

endmodule
[DONE]
