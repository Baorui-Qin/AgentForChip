module TopModule (
    input clk,
    input resetn,
    input x,
    input y,
    output reg f,
    output reg g
);

    localparam [3:0] A     = 4'd0,
                     F1    = 4'd1,
                     S0    = 4'd2,
                     S1    = 4'd3,
                     S2    = 4'd4,
                     G1    = 4'd5,
                     G2    = 4'd6,
                     GPERM = 4'd7,
                     GZERO = 4'd8;

    reg [3:0] state, next;

    always @(posedge clk) begin
        if (!resetn)
            state <= A;
        else
            state <= next;
    end

        f = 1'b0;
        g = 1'b0;
        next = state;

        case (state)

                if (resetn)
                    next = F1;
                else
                    next = A;
