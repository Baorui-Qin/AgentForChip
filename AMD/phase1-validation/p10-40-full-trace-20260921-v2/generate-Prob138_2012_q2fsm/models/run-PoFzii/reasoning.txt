    input  logic w,
    output logic z

    localparam logic [2:0] B = 3'd1;
    localparam logic [2:0] C = 3'd2;
    localparam logic [2:0] D = 3'd3;
    localparam logic [2:0] E = 3'd4;
    localparam logic [2:0] F = 3'd5;

    logic [2:0] state, next_state;

    always_comb begin
        case (state)
            A: next_state = w ? B : A;

            D: next_state = w ? F : A;
            E: next_state = w ? E : D;

        endcase
    end

        else

    end
