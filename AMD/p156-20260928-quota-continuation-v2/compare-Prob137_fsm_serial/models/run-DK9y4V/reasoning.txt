    output done
);

    reg [3:0] count;

    always @(posedge clk) begin
        if (reset) begin
            state <= IDLE;
            count <= 0;
        end
        else begin
            state <= next_state;
            if (state == IDLE) count <= 0;
            else if (state == DATA) count <= count + 1;

            IDLE: next_state = in ? IDLE : START;

            ERROR: next_state = in ? IDLE : ERROR;
            default: next_state = IDLE;
        endcase

endmodule
