module TopModule (
    input  clk,
    input  reset,
    input  s,
    input  w,
    output z
);

               B = 2'd1,
               C = 2'd2,
               D = 2'd3;

    reg [1:0] state, next_state;
    reg [1:0] count;

    always @(*) begin
        case (state)
            A:       next_state = s ? B : A;
            B:       next_state = C;
            C:       next_state = D;
            D:       next_state = B;
            default: next_state = A;
        endcase
    end

    always @(posedge clk) begin
        if (reset) begin
            state <= A;
            count <= 2'd0;
        end else begin
            state <= next_state;
            case (state)
                A:       count <= 2'd0;
                B:       count <= {1'b0, w};
                C:       count <= count + {1'b0, w};
                D:       count <= count + {1'b0, w};
                default: count <= 2'd0;
            endcase
        end
    end

    assign z = (state == B) && (count == 2'd2);

endmodule
