[BEGIN]
module TopModule (
    input clk,
    input reset,
    input data,
    output [3:0] count,
    output counting,
    output done,
    input ack
);

    localparam IDLE  = 2'd0;
    localparam DELAY = 2'd1;
    localparam COUNT = 2'd2;
    localparam DONE  = 2'd3;

    reg [1:0] state;
    reg [3:0] sr;
    reg [3:0] delay_reg;
    reg [1:0] bitcnt;
    reg [3:0] count_reg;
    reg [9:0] timer;

    assign counting = (state == COUNT);
    assign done     = (state == DONE);
    assign count    = count_reg;

    always @(posedge clk) begin
        if (reset) begin
            state     <= IDLE;
            sr        <= 4'b0;
            delay_reg <= 4'b0;
            bitcnt    <= 2'd0;
            count_reg <= 4'b0;
            timer     <= 10'd0;
        end else begin
            case (state)
                IDLE: begin
                    sr <= {sr[2:0], data};
                    if ({sr[2:0], data} == 4'b1101) begin
                        state     <= DELAY;
                        bitcnt    <= 2'd0;
                        delay_reg <= 4'b0;
                    end
                end

                DELAY: begin
                    delay_reg <= {delay_reg[2:0], data};
                    if (bitcnt == 2'd3) begin
                        state     <= COUNT;
                        count_reg <= {delay_reg[2:0], data};
                        timer     <= 10'd999;
                    end else begin
                        bitcnt <= bitcnt + 1'b1;
                    end
                end

                COUNT: begin
                    if (timer == 10'd0) begin
                        if (count_reg == 4'd0) begin
                            state <= DONE;
                        end else begin
                            count_reg <= count_reg - 1'b1;
                            timer     <= 10'd999;
                        end
                    end else begin
                        timer <= timer - 1'b1;
                    end
                end

                DONE: begin
                    if (ack) begin
                        state <= IDLE;
                        sr    <= 4'b0;
                    end
                end

                default: state <= IDLE;
            endcase
        end
    end

endmodule
[DONE]
