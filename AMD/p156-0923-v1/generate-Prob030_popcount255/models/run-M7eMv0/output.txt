[BEGIN]
module TopModule (
  input  logic [254:0] in,
  output logic [7:0] out
);

  logic [255:0] padded;
  assign padded = {1'b0, in};

  logic [1:0] s1 [0:127];
  logic [2:0] s2 [0:63];
  logic [3:0] s3 [0:31];
  logic [4:0] s4 [0:15];
  logic [5:0] s5 [0:7];
  logic [6:0] s6 [0:3];
  logic [7:0] s7 [0:1];

  genvar i;
  generate
    for (i = 0; i < 128; i = i + 1) begin : gen_s1
      assign s1[i] = padded[2*i] + padded[2*i+1];
    end
    for (i = 0; i < 64; i = i + 1) begin : gen_s2
      assign s2[i] = s1[2*i] + s1[2*i+1];
    end
    for (i = 0; i < 32; i = i + 1) begin : gen_s3
      assign s3[i] = s2[2*i] + s2[2*i+1];
    end
    for (i = 0; i < 16; i = i + 1) begin : gen_s4
      assign s4[i] = s3[2*i] + s3[2*i+1];
    end
    for (i = 0; i < 8; i = i + 1) begin : gen_s5
      assign s5[i] = s4[2*i] + s4[2*i+1];
    end
    for (i = 0; i < 4; i = i + 1) begin : gen_s6
      assign s6[i] = s5[2*i] + s5[2*i+1];
    end
    for (i = 0; i < 2; i = i + 1) begin : gen_s7
      assign s7[i] = s6[2*i] + s6[2*i+1];
    end
  endgenerate

  assign out = s7[0] + s7[1];

endmodule
[DONE]
