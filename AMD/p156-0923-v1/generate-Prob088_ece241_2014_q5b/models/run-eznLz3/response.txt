[BEGIN]
module TopModule (
    input  logic clk,
    input  logic areset,
    input  logic x,
    output logic z
);

    localparam logic [1:0] A = 2'b01;
    localparam logic [1:0] B = 2'b10;

    logic [1:0] state;
    logic [1:0] next;

    always_ff @(posedge clk or posedge areset) begin
        if (areset)
            state <= A;
        else
            state <= next;
    end

    always_comb begin
        case (state)
            A: next = x ? B : A;
            B: next = B;
            default: next = A;
        endcase
    end

    always_comb begin
        z = (state == A) ? x : ~x;
    end

endmodule
[DONE]
