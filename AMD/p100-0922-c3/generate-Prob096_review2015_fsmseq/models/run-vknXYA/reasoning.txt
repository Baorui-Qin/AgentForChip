module TopModule (
  input clk,
  input reset,
  input data,
  output start_shifting
);

  localparam S0=2'd0, S1=2'd1, S2=2'd2, S3=2'd3, S4=2'd4; need 3 bits. Use 3'd0..4.

  reg [2:0] state, next;

  always @(posedge clk) begin
    if (reset) state <= S0;
    else state <= next;
  end

  always @(*) begin
    next = state;
    case (state)
      S0: next = data ? S1 : S0;
      S1: next = data ? S2 : S0;
      S2: next = data ? S2 : S3;
      S3: next = data ? S4 : S0;
      S4: next = S4;
      default: next = S0;
    endcase
  end

  assign start_shifting = (state == S4);

endmodule
