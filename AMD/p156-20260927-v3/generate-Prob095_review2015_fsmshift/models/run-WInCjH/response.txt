[BEGIN]
module TopModule (
    input  clk,
    input  reset,
    output logic shift_ena
);

    logic [2:0] count;

    always_ff @(posedge clk) begin
        if (reset) begin
            count <= 3'd4;
        end else if (count != 3'd0) begin
            count <= count - 3'd1;
        end
    end

    assign shift_ena = (count != 3'd0);

endmodule
[DONE]
