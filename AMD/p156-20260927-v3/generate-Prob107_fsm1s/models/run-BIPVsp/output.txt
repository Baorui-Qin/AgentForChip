[BEGIN]
module TopModule (
    input  logic clk,
    input  logic reset,
    input  logic in,
    output logic out
);
    logic state;

    always_ff @(posedge clk) begin
        if (reset)
            state <= 1'b1;
        else
            state <= ~(in ^ state);
    end

    assign out = state;
endmodule
[DONE]
