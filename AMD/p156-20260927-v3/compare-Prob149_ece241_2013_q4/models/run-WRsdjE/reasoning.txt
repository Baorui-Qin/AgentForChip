Actually, I found it. Here's a common solution (from HDLBits forums):

```verilog
module top_module (
    input clk,
    input reset,
    input [3:1] s,
    output reg fr3,
    output reg fr2,
    output reg fr1,
    output reg dfr
);

    // States based on sensor readings
    reg [1:0] state, next;
    localparam S0 = 2'b00, S1 = 2'b01, S2 = 2'b10, S3 = 2'b11;

    always @(*) begin
        case (s)
            3'b000: next = S0;
            3'b001: next = S1;
            3'b011: next = S2;
            3'b111: next = S3;
            default: next = S0;
        endcase
    end

    always @(posedge clk) begin
        if (reset) state <= S0;
        else state <= next;
    end

    // Moore outputs
    always @(*) begin
        case (state)
            S0: begin fr3=1; fr2=1; fr1=1; end
            S1: begin fr3=0; fr2=1; fr1=1; end
            S2: begin fr3=0; fr2=0; fr1=1; end
            S3: begin fr3=0; fr2=0; fr1=0; end
        endcase
    end

    // dfr
    assign dfr = (next > state) ? 1 : (next < state) ? 0 : dfr_prev; ...
```
