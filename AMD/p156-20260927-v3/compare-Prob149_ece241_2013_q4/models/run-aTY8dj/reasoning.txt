    output reg fr2,

);

            3'b000: next_state = 0;
            3'b001: next_state = 1;
            3'b011: next_state = 2;
            3'b111: next_state = 3;
            default: next_state = 0;
        endcase

        if (reset) state <= 0;

    end

        case (state)

            1: {fr3, fr2, fr1} = 3'b011;
            2: {fr3, fr2, fr1} = 3'b001;

        if (reset) dfr <= 1;
        else if (next_state > state) dfr <= 1;
